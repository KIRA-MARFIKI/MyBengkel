export function isGeneratedCodeValid(
  expectedCode: string,
  enteredCode: string,
  nowMs: number = Date.now()
): boolean {
  if (!expectedCode || !enteredCode) return false;
  if (expectedCode.trim() !== enteredCode.trim()) return false;
  const expiresAt = codeExpiryStore.get(expectedCode);
  if (expiresAt !== undefined && nowMs > expiresAt) {
    return false;
  }
  return true;
}

export function maskPhone(hp: string): string {

}

export function maskEmail(email: string): string {
  const clean = email.trim();
  const atIdx = clean.indexOf('@');
  if (atIdx <= 1) return clean;
  const local = clean.slice(0, atIdx);
  const domain = clean.slice(atIdx);
  return `${local.slice(0, 2)}••••${domain}`;
}

const loadStoredSecuritySetup = (userEmail?: string, userRole?: UserRole): SecuritySetupProgress => {
  const isPrivileged =
    userRole === 'admin' || userRole === 'master_admin' || userRole === 'master_helpdesk';
  if (!isPrivileged) {
    return {
      mustCompleteSetup: false,
      passwordCompleted: true,
      emailCompleted: true,
      pinCompleted: true,
      completedCount: 3,
      totalSteps: 3,
    };
  }

  if (typeof localStorage === 'undefined' || !userEmail) {
    return {
      mustCompleteSetup: true,
      passwordCompleted: false,
      emailCompleted: false,
      pinCompleted: false,
      completedCount: 0,
      totalSteps: 3,
    };
  }

  const cleanEmail = userEmail.trim().toLowerCase();
  const isDefaultBootstrap = DEFAULT_BOOTSTRAP_EMAILS.includes(cleanEmail);

  try {
    const raw = localStorage.getItem(`${SECURITY_SETUP_STORAGE_KEY}_${cleanEmail}`);
    if (raw) {
      const parsed = JSON.parse(raw);
      const pw = Boolean(parsed.passwordCompleted);
      const em = Boolean(parsed.emailCompleted);
      const pin = Boolean(parsed.pinCompleted);
      const count = (pw ? 1 : 0) + (em ? 1 : 0) + (pin ? 1 : 0);
      // Bootstrap accounts must ALWAYS complete setup by binding a personalized email & PIN
      const must = isDefaultBootstrap ? true : count < 3;
      return {
        mustCompleteSetup: must,
        passwordCompleted: pw,
        emailCompleted: em,
        pinCompleted: pin,
        completedCount: count,
        totalSteps: 3,
      };
    }
  } catch {
    // safe fallback
  }

  return {
    mustCompleteSetup: true,
    passwordCompleted: false,
    emailCompleted: false,
    pinCompleted: false,
    completedCount: 0,
    totalSteps: 3,
  };
};

          },

      }
    } catch (err: any) {
      console.warn('Supabase signUp network exception:', err);
      requestError = String(err?.message || err || 'Network error');
    }

      const lower = requestError.toLowerCase();
      if (

    const next: SecuritySetupProgress = {
      ...securitySetup,
      emailCompleted: false,
      completedCount: nextCount,

// Route guard: requireRole("admin" | "master_admin") for every regular operational /admin route
export const RequireAdmin: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const { isLoggedIn, role } = useAuth();
  const location = useLocation();

  if (!isLoggedIn) {
    return <Navigate to={`/login?next=${encodeURIComponent(location.pathname)}`} replace />;
  }

  // If a Master Helpdesk attempts to access non-helpdesk admin operations, redirect to Helpdesk Center
  if (role === 'master_helpdesk') {
    return <Navigate to="/admin/helpdesk" replace />;
  }

  if (role !== 'admin' && role !== 'master_admin') {
    return <Navigate to={`/login?next=${encodeURIComponent(location.pathname)}`} replace />;
  }

  return <>{children}</>;
};

// Route guard: allow "admin", "master_admin", and "master_helpdesk" for Helpdesk Center
export const RequireHelpdeskOrAdmin: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const { isLoggedIn, role } = useAuth();
  const location = useLocation();

    return <Navigate to={`/login?next=${encodeURIComponent(location.pathname)}`} replace />;
  }

  return <>{children}</>;
};

    e.preventDefault();
    setPasswordError('');
    setPasswordNotice('');

    if (passwordStrengthScore < 3 || !hasMinLength) {
      setPasswordError(
        'Kata sandi baru minimal 8 karakter dan mengandung kombinasi huruf besar, huruf kecil, angka, serta simbol.'
      );
      return;
    }

    if (['admin123', 'masteradmin123', 'helpdesk123'].includes(newPassword.trim())) {
      setPasswordError('Kata sandi baru tidak boleh sama dengan kata sandi bawaan.');
      return;
    }

    if (newPassword !== confirmPassword) {
      setPasswordError('Konfirmasi kata sandi baru tidak sesuai.');
      return;
    }

    setPasswordLoading(true);
    try {
      const res = await updateSecurityPassword(newPassword);
      if (!res.ok) {
        setPasswordError(res.error || 'Gagal memperbarui kata sandi.');
      } else {
        setPasswordNotice('Kata sandi telah diperbarui.');
        setNewPassword('');
        setConfirmPassword('');
        setCurrentPassword('');
        // Auto-switch to next incomplete step (Email)
        if (!securitySetup.emailCompleted) {
          setActiveTab('email');
        } else if (!securitySetup.pinCompleted) {
          setActiveTab('pin');
        }
      }
    } finally {
      setPasswordLoading(false);
    }
  };

  // Submit Email Form
  const handleSubmitEmail = async (e: React.FormEvent) => {
    e.preventDefault();
    setEmailError('');
    setEmailNotice('');

    const cleanNew = newEmail.trim().toLowerCase();
    const cleanConfirm = confirmEmail.trim().toLowerCase();
    const current = (user?.email || '').trim().toLowerCase();

    if (!cleanNew) {
      setEmailError('Email baru wajib diisi.');
      return;
    }

    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(cleanNew)) {
      setEmailError('Format email baru belum valid.');
      return;
    }

    if (cleanNew === current) {
      setEmailError('Email baru harus berbeda dengan email saat ini.');
      return;
    }

    if (DEFAULT_BOOTSTRAP_EMAILS.includes(cleanNew)) {
      setEmailError('Email baru tidak boleh menggunakan alamat email bawaan sistem.');
      return;
    }

    if (cleanNew !== cleanConfirm) {
      setEmailError('Konfirmasi email baru tidak sesuai.');
      return;
    }

    setEmailLoading(true);
    try {
      const res = await updateSecurityEmail(cleanNew);
      if (!res.ok) {
        setEmailError(res.error || 'Gagal mengirim permintaan verifikasi email.');
      } else {
        setEmailNotice(
          `Tautan verifikasi telah dikirim ke ${cleanNew}. Silakan periksa kotak masuk email Anda dan klik tautan untuk mengonfirmasi akun.`
        );
        setResendCooldown(60);
      }
    } finally {
      setEmailLoading(false);
    }
  };

  // Submit PIN Form
  const handleSubmitPin = async (e: React.FormEvent) => {
    e.preventDefault();
    setPinError('');
    setPinNotice('');

    const cleanPin = newPin.trim();
    if (cleanPin.length !== 12 || !/^\d{12}$/.test(cleanPin)) {
      setPinError('PIN baru wajib terdiri dari 12 angka.');
      return;
    }

    if (cleanPin !== confirmPin.trim()) {
      setPinError('Konfirmasi PIN baru tidak sesuai.');
      return;
    }

    setPinLoading(true);
    try {
      const res = await updateSecurityPin(cleanPin);
      if (!res.ok) {
        setPinError(res.error || 'Gagal menyimpan PIN baru.');
      } else {
        setPinNotice('PIN baru telah disimpan. Keamanan akun lengkap.');
        setNewPin('');
        setConfirmPin('');

        // Auto-redirect immediately to destination dashboard
        const dest = role === 'master_helpdesk' ? '/admin/helpdesk' : '/admin';
        navigate(dest, { replace: true });
      }
    } finally {
      setPinLoading(false);
    }
  };

  return (
    <div
      role="dialog"
      aria-modal="true"
      aria-label="Keamanan Akun"
      className="fixed inset-0 z-[9999] flex items-center justify-center bg-black/70 backdrop-blur-md p-3 sm:p-4 overflow-y-auto"
    >
      <div className="relative w-full max-w-[620px] bg-white rounded-[16px] shadow-2xl border border-[#D9DDE6] overflow-hidden my-4 sm:my-6">
        {/* Header */}
        <div className="p-4 sm:p-6 md:p-8 bg-[#1E2230] text-white">
          <div className="flex items-center justify-between gap-3">
            <div className="flex flex-wrap items-center gap-2 min-w-0">
              <span className="inline-flex items-center px-2.5 py-1 rounded-full text-[11px] sm:text-xs font-semibold uppercase tracking-wider bg-[#FFBD59]/20 text-[#FFBD59] whitespace-nowrap">
                Wajib: Keamanan Akun
              </span>
              <span className="text-[11px] sm:text-xs text-[#9DA4B4] font-medium whitespace-nowrap">
                Peran: <strong className="text-white capitalize">{role?.replace('_', ' ')}</strong>
              </span>
            </div>

            {/* Mobile Close (X) icon button (< 768px) with minimum 44x44px touch target */}
            <button
              type="button"
              onClick={logout}
              aria-label="Keluar dari Sesi"
              title="Keluar dari Sesi"
              className="md:hidden flex items-center justify-center w-11 h-11 -mr-2 -my-2 rounded-full text-white/80 hover:text-white hover:bg-white/10 active:scale-95 transition-all shrink-0"
            >
              <svg width="20" height="20" viewBox="0 0 20 20" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
                <line x1="5" y1="5" x2="15" y2="15" />
                <line x1="15" y1="5" x2="5" y2="15" />
              </svg>
            </button>
          </div>

          {/* Progress Bar & Counter */}
          <div className="mt-4 sm:mt-5 pt-3 sm:pt-4 border-t border-white/10">
            <div className="flex items-center justify-between text-xs font-medium mb-1.5">
              <span>Status Verifikasi</span>
              <span className="tabular-nums font-semibold text-[#FFBD59]">
                {securitySetup.completedCount} dari {securitySetup.totalSteps} Selesai
              </span>
            </div>
            <div className="w-full h-2 rounded-full bg-white/10 overflow-hidden">
              <div
                className="h-full bg-[#FFBD59] transition-all duration-300"
                style={{
                  width: `${(securitySetup.completedCount / securitySetup.totalSteps) * 100}%`,
                }}
              />
            </div>
          </div>
        </div>

        {/* 3 Steps Tabs */}
        <div className="flex border-b border-[#EEF0F4] bg-[#F7F8FA]">
          <button
            type="button"
            onClick={() => setActiveTab('password')}
            className={`flex-1 min-h-[44px] py-2.5 sm:py-3 px-1.5 sm:px-2 text-center text-[11px] sm:text-xs font-medium border-b-2 transition-colors flex items-center justify-center gap-1 sm:gap-1.5 ${
              activeTab === 'password'
                ? 'border-[#1E2230] text-[#1E2230] font-semibold bg-white'
                : 'border-transparent text-[#646B7C] hover:text-[#1E2230]'
            }`}
          >
            <span
              className={`w-4 h-4 rounded-full text-[10px] flex items-center justify-center font-bold shrink-0 ${
                securitySetup.passwordCompleted
                  ? 'bg-[#22C55E] text-white'
                  : 'bg-[#D9DDE6] text-[#646B7C]'
              }`}
            >
              {securitySetup.passwordCompleted ? '✓' : '1'}
            </span>
            <span className="whitespace-nowrap font-medium">Password</span>
            <span className="hidden sm:inline text-[11px] opacity-75">
              ({securitySetup.passwordCompleted ? 'Selesai' : 'Belum'})
            </span>
          </button>

          <button
            type="button"
            onClick={() => setActiveTab('email')}
            className={`flex-1 min-h-[44px] py-2.5 sm:py-3 px-1.5 sm:px-2 text-center text-[11px] sm:text-xs font-medium border-b-2 transition-colors flex items-center justify-center gap-1 sm:gap-1.5 ${
              activeTab === 'email'
                ? 'border-[#1E2230] text-[#1E2230] font-semibold bg-white'
                : 'border-transparent text-[#646B7C] hover:text-[#1E2230]'
            }`}
          >
            <span
              className={`w-4 h-4 rounded-full text-[10px] flex items-center justify-center font-bold shrink-0 ${
                securitySetup.emailCompleted
                  ? 'bg-[#22C55E] text-white'
                  : 'bg-[#D9DDE6] text-[#646B7C]'
              }`}
            >
              {securitySetup.emailCompleted ? '✓' : '2'}
            </span>
            <span className="whitespace-nowrap font-medium">Email Akun</span>
            <span className="hidden sm:inline text-[11px] opacity-75">
              ({securitySetup.emailCompleted ? 'Selesai' : 'Menunggu'})
            </span>
          </button>

          <button
            type="button"
            disabled={!securitySetup.emailCompleted}
            onClick={() => {
              if (securitySetup.emailCompleted) {
                setActiveTab('pin');
              }
            }}
            title={
              !securitySetup.emailCompleted
                ? 'Selesaikan konfirmasi email terlebih dahulu sebelum membuat PIN baru'
                : 'Buat PIN 12 digit'
            }
            className={`flex-1 min-h-[44px] py-2.5 sm:py-3 px-1.5 sm:px-2 text-center text-[11px] sm:text-xs font-medium border-b-2 transition-colors flex items-center justify-center gap-1 sm:gap-1.5 ${
              !securitySetup.emailCompleted
                ? 'opacity-50 cursor-not-allowed border-transparent text-[#9DA4B4] bg-[#F7F8FA]'
                : activeTab === 'pin'
                ? 'border-[#1E2230] text-[#1E2230] font-semibold bg-white'
                : 'border-transparent text-[#646B7C] hover:text-[#1E2230]'
            }`}
          >
            <span
              className={`w-4 h-4 rounded-full text-[10px] flex items-center justify-center font-bold shrink-0 ${
                securitySetup.pinCompleted
                  ? 'bg-[#22C55E] text-white'
                  : 'bg-[#D9DDE6] text-[#646B7C]'
              }`}
            >
              {securitySetup.pinCompleted ? '✓' : '3'}
            </span>
            <span className="whitespace-nowrap font-medium">PIN Baru</span>
            <span className="hidden sm:inline text-[11px] opacity-75">
              ({securitySetup.pinCompleted ? 'Selesai' : !securitySetup.emailCompleted ? 'Terkunci' : 'Belum'})
            </span>
          </button>
        </div>

        {/* Content Section */}
        <div className="p-4 sm:p-6 md:p-8">
          {/* TAB 1: CHANGE PASSWORD */}
          {activeTab === 'password' && (
            <form onSubmit={handleSubmitPassword} className="space-y-4">
              <div className="flex items-center justify-between pb-2 border-b border-[#EEF0F4]">
                <div>
                  <h3 className="text-sm font-semibold text-[#1E2230]">Ganti Kata Sandi</h3>
                  <p className="text-xs text-[#646B7C]">
                    Buat kata sandi baru yang kuat untuk menggantikan kata sandi bawaan.
                  </p>
                </div>
                {securitySetup.passwordCompleted && (
                  <span className="px-2 py-0.5 rounded text-[11px] font-semibold bg-[#22C55E]/10 text-[#16A34A]">
                    Selesai
                  </span>
                )}
              </div>

              {/* Current Password Field */}
              <div>
                <label
                  htmlFor="setup-current-password"
                  className="block text-xs font-semibold text-[#1E2230] mb-1"
                >
                  Kata Sandi Saat Ini (Opsional untuk verifikasi)
                </label>
                <input
                  id="setup-current-password"
                  type={showPassword ? 'text' : 'password'}
                  value={currentPassword}
                  onChange={(e) => setCurrentPassword(e.target.value)}
                  placeholder="Masukkan kata sandi saat ini"
                  className="w-full px-3 py-2 text-sm border border-[#D9DDE6] rounded-[6px] focus:outline-none focus:border-[#1E2230]"
                />
              </div>

              {/* New Password Field */}
              <div>
                <div className="flex items-center justify-between mb-1">
                  <label
                    htmlFor="setup-new-password"
                    className="block text-xs font-semibold text-[#1E2230]"
                  >
                    Kata Sandi Baru
                  </label>
                  <button
                    type="button"
                    onClick={() => setShowPassword((s) => !s)}
                    className="text-xs text-[#646B7C] hover:text-[#1E2230] inline-flex items-center gap-1"
                  >
                    {showPassword ? <EyeOffIcon size={14} /> : <EyeIcon size={14} />}
                    <span>{showPassword ? 'Sembunyikan' : 'Tampilkan'}</span>
                  </button>
                </div>
                <input
                  id="setup-new-password"
                  type={showPassword ? 'text' : 'password'}
                  value={newPassword}
                  onChange={(e) => setNewPassword(e.target.value)}
                  placeholder="Minimal 8 karakter campuran"
                  className="w-full px-3 py-2 text-sm border border-[#D9DDE6] rounded-[6px] focus:outline-none focus:border-[#1E2230]"
                />

                {/* Password Strength Meter */}
                <div className="mt-2 space-y-1.5">
                  <div className="flex items-center justify-between text-[11px]">
                    <span className="text-[#646B7C]">Kekuatan kata sandi:</span>
                    <span className="font-semibold text-[#1E2230]">{getStrengthLabel()}</span>
                  </div>
                  <div className="w-full h-1.5 bg-[#EEF0F4] rounded-full overflow-hidden">
                    <div
                      className={`h-full ${getStrengthColor()} transition-all`}
                      style={{ width: `${(passwordStrengthScore / 5) * 100}%` }}
                    />
                  </div>
                  <div className="grid grid-cols-2 gap-1 text-[11px] text-[#646B7C] pt-1">
                    <span className={hasMinLength ? 'text-[#16A34A] font-medium' : ''}>
                      {hasMinLength ? '✓' : '•'} Min. 8 karakter
                    </span>
                    <span className={hasUpper ? 'text-[#16A34A] font-medium' : ''}>
                      {hasUpper ? '✓' : '•'} Huruf besar
                    </span>
                    <span className={hasLower ? 'text-[#16A34A] font-medium' : ''}>
                      {hasLower ? '✓' : '•'} Huruf kecil
                    </span>
                    <span className={hasDigit && hasSpecial ? 'text-[#16A34A] font-medium' : ''}>
                      {hasDigit && hasSpecial ? '✓' : '•'} Angka & simbol
                    </span>
                  </div>
                </div>
              </div>

              {/* Confirm New Password */}
              <div>
                <label
                  htmlFor="setup-confirm-password"
                  className="block text-xs font-semibold text-[#1E2230] mb-1"
                >
                  Konfirmasi Kata Sandi Baru
                </label>
                <input
                  id="setup-confirm-password"
                  type={showPassword ? 'text' : 'password'}
                  value={confirmPassword}
                  onChange={(e) => setConfirmPassword(e.target.value)}
                  placeholder="Ketik ulang kata sandi baru"
                  className="w-full px-3 py-2 text-sm border border-[#D9DDE6] rounded-[6px] focus:outline-none focus:border-[#1E2230]"
                />
              </div>

              {passwordError && (
                <div className="p-3 rounded-[6px] bg-[#FFF2F5] border border-[#FAD2DC] text-xs text-[#D62E5A]">
                  {passwordError}
                </div>
              )}

              {passwordNotice && (
                <div className="p-3 rounded-[6px] bg-[#F0FDF4] border border-[#BBF7D0] text-xs text-[#16A34A]">
                  {passwordNotice}
                </div>
              )}

              <div className="pt-2 flex flex-col sm:flex-row items-stretch sm:items-center justify-end">
                <ActionButton
                  type="submit"
                  intent="start"
                  disabled={passwordLoading}
                  className="w-full sm:w-auto justify-center min-h-[44px]"
                >
                  {passwordLoading ? 'Menyimpan...' : 'Simpan Kata Sandi'}
                </ActionButton>
              </div>
            </form>
          )}

          {/* TAB 2: CHANGE EMAIL */}
          {activeTab === 'email' && (
            <form onSubmit={handleSubmitEmail} className="space-y-4">
              <div className="flex items-center justify-between pb-2 border-b border-[#EEF0F4]">
                <div>
                  <h3 className="text-sm font-semibold text-[#1E2230]">Ganti Email Akun</h3>
                  <p className="text-xs text-[#646B7C]">
                    Perbarui email akun resmi Anda untuk mengaktifkan notifikasi dan pemulihan.
                  </p>
                </div>
                <span
                  className={`px-2 py-0.5 rounded text-[11px] font-semibold ${
                    securitySetup.emailCompleted
                      ? 'bg-[#22C55E]/10 text-[#16A34A]'
                      : 'bg-[#FFBD59]/20 text-[#B45309]'
                  }`}
                >
                  {securitySetup.emailCompleted ? 'Selesai' : 'Menunggu Konfirmasi'}
                </span>
              </div>

              {/* Current Email (Read-only) */}
              <div>
                <label className="block text-xs font-semibold text-[#646B7C] mb-1">
                  Email Akun Saat Ini (Terdaftar)
                </label>
                <input
                  type="text"
                  readOnly
                  disabled
                  value={user?.email || ''}
                  className="w-full px-3 py-2 text-sm bg-[#F6F7FA] border border-[#D9DDE6] rounded-[6px] text-[#646B7C] cursor-not-allowed font-mono"
                />
              </div>

              {/* New Email */}
              <div>
                <label
                  htmlFor="setup-new-email"
                  className="block text-xs font-semibold text-[#1E2230] mb-1"
                >
                  Email Baru
                </label>
                <input
                  id="setup-new-email"
                  type="email"
                  value={newEmail}
                  onChange={(e) => setNewEmail(e.target.value)}
                  placeholder="nama@domain-anda.com"
                  className="w-full px-3 py-2 text-sm border border-[#D9DDE6] rounded-[6px] focus:outline-none focus:border-[#1E2230]"
                />
              </div>

              {/* Confirm New Email */}
              <div>
                <label
                  htmlFor="setup-confirm-email"
                  className="block text-xs font-semibold text-[#1E2230] mb-1"
                >
                  Konfirmasi Email Baru
                </label>
                <input
                  id="setup-confirm-email"
                  type="email"
                  value={confirmEmail}
                  onChange={(e) => setConfirmEmail(e.target.value)}
                  placeholder="Ketik ulang email baru"
                  className="w-full px-3 py-2 text-sm border border-[#D9DDE6] rounded-[6px] focus:outline-none focus:border-[#1E2230]"
                />
              </div>

              <div className="p-3 rounded-[6px] bg-[#F7F8FA] border border-[#EEF0F4] text-xs text-[#646B7C] space-y-1">
                <p className="font-semibold text-[#1E2230]">Catatan Konfirmasi Email:</p>
                <p>
                  Sesuai standar keamanan Supabase Auth, tautan verifikasi akan dikirimkan. Langkah 3
                  (Pembuatan PIN Baru) akan terbuka secara otomatis setelah Anda mengonfirmasi email
                  melalui tautan yang dikirim.
                </p>
              </div>

              {emailError && (
                <div className="space-y-2">
                  <div className="p-3 rounded-[6px] bg-[#FFF2F5] border border-[#FAD2DC] text-xs text-[#D62E5A] leading-relaxed">
                    {emailError}
                  </div>
                  <div className="p-3 rounded-[6px] bg-[#F7F8FA] border border-[#EEF0F4] text-[11px] text-[#646B7C] flex flex-wrap items-center justify-between gap-2">
                    <span>Sedang menguji lokal tanpa SMTP aktif?</span>
                    <Link
                      to={`/auth/verify-email?role=${role}`}
                      className="font-semibold text-[#1E2230] underline hover:text-[#2C3246]"
                    >
                      Lanjutkan dengan Simulasi Verifikasi
                    </Link>
                  </div>
                </div>
              )}

              {emailNotice && (
                <div className="p-4 rounded-[8px] bg-[#F0FDF4] border border-[#BBF7D0] text-xs text-[#16A34A] space-y-2">
                  <div className="font-medium">{emailNotice}</div>
                  <div className="pt-1 flex flex-wrap items-center justify-between gap-2 border-t border-[#BBF7D0]/50 text-[11px]">
                    <span className="text-[#646B7C]">
                      Belum menerima email? Tunggu{' '}
                      {resendCooldown > 0 ? (
                        <strong className="text-[#1E2230]">{resendCooldown} detik</strong>
                      ) : (
                        'sebentar'
                      )}
                    </span>
                    <Link
                      to={`/auth/verify-email?role=${role}`}
                      className="font-semibold text-[#1E2230] underline hover:text-[#2C3246]"
                    >
                      Buka Konfirmasi Email Langsung (Simulasi)
                    </Link>
                  </div>
                </div>
              )}

              <div className="pt-2 flex flex-col sm:flex-row items-stretch sm:items-center justify-between gap-3">
                {resendCooldown > 0 ? (
                  <span className="text-xs text-[#646B7C] text-center sm:text-left">
                    Kirim ulang dalam <strong className="tabular-nums">{resendCooldown}s</strong>
                  </span>
                ) : (
                  <span className="hidden sm:inline" />
                )}
                <ActionButton
                  type="submit"
                  intent="start"
                  disabled={emailLoading || resendCooldown > 0}
                  className="w-full sm:w-auto justify-center min-h-[44px]"
                >
                  {emailLoading
                    ? 'Mengirim...'
                    : resendCooldown > 0
                    ? `Kirim Ulang (${resendCooldown}s)`
                    : 'Kirim Tautan Verifikasi Email'}
                </ActionButton>
              </div>
            </form>
          )}

          {/* TAB 3: CREATE NEW PIN */}
          {activeTab === 'pin' && (
            <form onSubmit={handleSubmitPin} className="space-y-4">
              <div className="flex items-center justify-between pb-2 border-b border-[#EEF0F4]">
                <div>
                  <h3 className="text-sm font-semibold text-[#1E2230]">Buat PIN Baru</h3>
                  <p className="text-xs text-[#646B7C]">
                    Tetapkan PIN rahasia 12 angka untuk otorisasi tindakan sensitif.
                  </p>
                </div>
                {securitySetup.pinCompleted && (
                  <span className="px-2 py-0.5 rounded text-[11px] font-semibold bg-[#22C55E]/10 text-[#16A34A]">
                    Selesai
                  </span>
                )}
              </div>

              {/* New PIN */}
              <div>
                <div className="flex items-center justify-between mb-1">
                  <label
                    htmlFor="setup-new-pin"
                    className="block text-xs font-semibold text-[#1E2230]"
                  >
                    PIN Baru (12 Digit Angka)
                  </label>
                  <button
                    type="button"
                    onClick={() => setShowPin((s) => !s)}
                    className="text-xs text-[#646B7C] hover:text-[#1E2230] inline-flex items-center gap-1"
                  >
                    {showPin ? <EyeOffIcon size={14} /> : <EyeIcon size={14} />}
                    <span>{showPin ? 'Sembunyikan' : 'Tampilkan'}</span>
                  </button>
                </div>
                <input
                  id="setup-new-pin"
                  type={showPin ? 'text' : 'password'}
                  inputMode="numeric"
                  maxLength={12}
                  value={newPin}
                  onChange={(e) => setNewPin(e.target.value.replace(/\D/g, ''))}
                  placeholder="123456789012"
                  className="w-full px-3 py-2 text-sm border border-[#D9DDE6] rounded-[6px] font-mono tracking-widest focus:outline-none focus:border-[#1E2230]"
                />
                <span className="block text-[11px] text-[#646B7C] mt-1">
                  Panjang saat ini: <strong className="tabular-nums">{newPin.length}</strong> / 12 digit angka.
                </span>
              </div>

              {/* Confirm New PIN */}
              <div>
                <label
                  htmlFor="setup-confirm-pin"
                  className="block text-xs font-semibold text-[#1E2230] mb-1"
                >
                  Konfirmasi PIN Baru
                </label>
                <input
                  id="setup-confirm-pin"
                  type={showPin ? 'text' : 'password'}
                  inputMode="numeric"
                  maxLength={12}
                  value={confirmPin}

-- 6. RPC: SAFE ATOMIC APPROVAL OF DATA CORRECTION (Backend-Verified)
create or replace function public.approve_data_correction(
  p_correction_id text,
  p_decision text,
  p_review_notes text
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_req record;
  v_reviewer_id uuid;
  v_reviewer_role text;
begin
  v_reviewer_id := auth.uid();
  v_reviewer_role := public.get_auth_role(v_reviewer_id);

  -- Enforce that Master Helpdesk cannot self-approve corrections
  if v_reviewer_role <> 'admin' and v_reviewer_role <> 'master_gudang' then
    raise exception 'Akses ditolak: Hanya peran Admin / Master Admin yang berwenang memberikan otorisasi koreksi.';
  end if;

  select * into v_req
  from public.helpdesk_correction_requests
  where id = p_correction_id for update;

  if not found then
    raise exception 'Permohonan koreksi tidak ditemukan.';
  end if;

  if v_req.status <> 'menunggu_persetujuan' then
    raise exception 'Permohonan ini telah diproses sebelumnya.';
  end if;

  -- Apply update to correction request record
  update public.helpdesk_correction_requests
  set status = p_decision,
      reviewed_by = (select coalesce(email, 'Admin') from auth.users where id = v_reviewer_id),
      reviewed_at = to_char(now(), 'YYYY-MM-DD HH24:MI'),
      review_notes = p_review_notes
  where id = p_correction_id;

  -- If approved, apply atomic modification to the target table according to entity_type
  if p_decision = 'disetujui' then
    if v_req.entity_type = 'Part' and v_req.field_name = 'stok_min' then
      update public.parts
      set stok_min = v_req.proposed_value::integer
      where id = v_req.entity_id or kode = v_req.entity_id;
    elsif v_req.entity_type = 'Booking' and v_req.field_name = 'status' then
      update public.bookings
      set status = v_req.proposed_value
      where id = v_req.entity_id;
    end if;
  end if;

  return jsonb_build_object(

-- 7. RPC: Set Account Security PIN

declare

-- 9. RPC: Verify and Reset PIN
-- Validates single-use token and updates user PIN securely
create or replace function public.verify_and_reset_pin(p_token text, p_new_pin text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, auth
as $$
declare
  v_clean_token text := trim(coalesce(p_token, ''));
  v_clean_pin text := trim(coalesce(p_new_pin, ''));
  v_rec record;
  v_role text;
  v_found boolean := false;
  v_new_pin_hash text;
begin
  if v_clean_token = '' then
    return jsonb_build_object('ok', false, 'error', 'Token pemulihan PIN tidak valid.');
  end if;

  -- Find active, unexpired, unused token
  for v_rec in
    select id, user_id, email, token_hash
    from public.pin_recovery_tokens
    where used_at is null and expires_at > now()
    order by created_at desc
  loop
    if v_rec.token_hash = extensions.crypt(v_clean_token, v_rec.token_hash) then
      v_found := true;
      exit;
    end if;
  end loop;

  if not v_found then
    return jsonb_build_object('ok', false, 'error', 'Tautan pemulihan PIN tidak valid atau telah kedaluwarsa.');
  end if;

  -- Check user role
  select role into v_role from public.user_roles where user_id = v_rec.user_id;

  -- Validate PIN length and numeric format:
  -- Privileged accounts (admin, master_admin, master_helpdesk) strictly require exactly 12 numeric digits
  if v_role in ('admin', 'master_admin', 'master_helpdesk') then
    if length(v_clean_pin) != 12 or v_clean_pin !~ '^\d{12}$' then
      return jsonb_build_object('ok', false, 'error', 'PIN wajib terdiri dari 12 angka.');
    end if;
  else
    if length(v_clean_pin) < 8 or length(v_clean_pin) > 12 or v_clean_pin !~ '^\d+$' then
      return jsonb_build_object('ok', false, 'error', 'PIN baru harus berupa angka antara 8 hingga 12 digit.');
    end if;
  end if;

  -- Hash new PIN
  v_new_pin_hash := extensions.crypt(v_clean_pin, extensions.gen_salt('bf', 8));

  -- Update PIN in public.user_security
  insert into public.user_security (user_id, email, pin_hash, updated_at)
  values (v_rec.user_id, v_rec.email, v_new_pin_hash, now())
  on conflict (user_id) do update
  set pin_hash = v_new_pin_hash,
      updated_at = now();

  -- Update account_security
  insert into public.account_security (user_id, pin_created_at, updated_at)
  values (v_rec.user_id, now(), now())
  on conflict (user_id) do update
  set pin_created_at = now(),
      updated_at = now();

  -- Mark token as consumed atomically
  update public.pin_recovery_tokens
  set used_at = now()
  where id = v_rec.id;

  return jsonb_build_object(
    'ok', true,
    'message', 'PIN telah diperbarui.'
  );
end;
$$;

-- 3a. Optional profile sync. public.profiles is not part of MyBengkel migrations,
--     so this helper only acts when the table and the needed columns exist and it
--     never raises (a profile problem must not block PIN setup).
create or replace function public.mb_sync_profile(p_uid uuid, p_email text, p_name text)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_has_id boolean;
  v_has_email boolean;
  v_has_name boolean;
  v_exists boolean;
begin
  if to_regclass('public.profiles') is null then
    return;
  end if;

  select
    bool_or(column_name = 'id'),
    bool_or(column_name = 'email'),
    bool_or(column_name = 'full_name')
  into v_has_id, v_has_email, v_has_name
  from information_schema.columns
  where table_schema = 'public' and table_name = 'profiles';

  if not coalesce(v_has_id, false) then
    return;
  end if;

  execute 'select exists (select 1 from public.profiles where id = $1)'
    into v_exists using p_uid;

  if not v_exists and coalesce(v_has_email, false) and p_email is not null then
    execute 'select exists (select 1 from public.profiles where lower(email) = lower($1))'
      into v_exists using p_email;
    if v_exists then
      return; -- a profile with this email already belongs to another row, do not touch it
    end if;
  end if;

  if v_exists then
    if coalesce(v_has_email, false) then
      execute 'update public.profiles set email = $2 where id = $1' using p_uid, p_email;
    end if;
  else
    if coalesce(v_has_email, false) and coalesce(v_has_name, false) then
      execute 'insert into public.profiles (id, email, full_name) values ($1, $2, $3)'
        using p_uid, p_email, p_name;
    elsif coalesce(v_has_email, false) then
      execute 'insert into public.profiles (id, email) values ($1, $2)' using p_uid, p_email;
    else
      execute 'insert into public.profiles (id) values ($1)' using p_uid;
    end if;
  end if;
exception
  when others then
    raise notice 'mb_sync_profile skipped: %', sqlerrm;
end;
$$;

-- 3b. Resolve the role of a user. Keeps the existing authoritative role row;
--     only inserts when no role row exists (from onboarding metadata).
create or replace function public.mb_ensure_user_role(p_uid uuid, p_fallback_role text)
returns text
language plpgsql
security definer
set search_path = public, auth
as $$
declare
  v_role text;
  v_meta_role text;
begin
  select role into v_role
  from public.user_roles
  where user_id = p_uid
  order by assigned_at desc nulls last
  limit 1;

  if v_role is not null then
    return v_role;
  end if;

  select raw_user_meta_data->>'onboarding_role' into v_meta_role
  from auth.users
  where id = p_uid;

  if v_meta_role in ('admin', 'master_admin', 'master_helpdesk', 'admin_gudang', 'master_gudang', 'customer') then
    v_role := v_meta_role;
  else
    v_role := p_fallback_role;
  end if;

  if v_role is null then
    return null;
  end if;

  if not exists (select 1 from public.user_roles where user_id = p_uid and role = v_role) then
    if exists (select 1 from public.user_roles where user_id = p_uid) then
      update public.user_roles set role = v_role, assigned_at = now() where user_id = p_uid;
    else
      insert into public.user_roles (user_id, role) values (p_uid, v_role);
    end if;
  end if;

  return v_role;
end;
$$;

-- 4. RPC: get_account_security_status
-- ------------------------------------------------------------------------------
create or replace function public.get_account_security_status()
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, auth
as $$
declare
  v_uid uuid := auth.uid();
  v_user record;
  v_sec record;
  v_role text;
  v_is_privileged boolean := false;
  v_email_confirmed boolean := false;
  v_pw_done boolean := false;
  v_email_done boolean := false;
  v_pin_done boolean := false;
  v_must_complete boolean := false;
  v_completed_count int := 0;
begin
  if v_uid is null then
    return jsonb_build_object('ok', false, 'error', 'Sesi tidak terautentikasi.');
  end if;

  select id, email, email_confirmed_at into v_user from auth.users where id = v_uid;
  if not found then
    return jsonb_build_object('ok', false, 'error', 'Data pengguna auth tidak ditemukan.');
  end if;

  v_email_confirmed := v_user.email_confirmed_at is not null;
  v_role := public.mb_ensure_user_role(v_uid, null);
  v_is_privileged := coalesce(v_role in ('admin', 'master_admin', 'master_helpdesk'), false);

  perform public.mb_ensure_account_security(v_uid, v_is_privileged);

  -- Sync email verification from auth.users
  if v_email_confirmed then
    update public.account_security
    set email_verified_at = coalesce(email_verified_at, v_user.email_confirmed_at),
        updated_at = now()
    where user_id = v_uid and email_verified_at is null;
  end if;

  select must_complete_security_setup, password_changed_at, email_verified_at,
         pin_created_at, security_setup_completed_at
  into v_sec
  from public.account_security
  where user_id = v_uid
  order by updated_at desc nulls last
  limit 1;

  if v_is_privileged then
    v_pw_done := v_sec.password_changed_at is not null;
    v_email_done := v_sec.email_verified_at is not null;
    v_pin_done := v_sec.pin_created_at is not null;
    v_completed_count := (case when v_pw_done then 1 else 0 end)
                       + (case when v_email_done then 1 else 0 end)
                       + (case when v_pin_done then 1 else 0 end);

    if v_completed_count >= 3 and v_sec.must_complete_security_setup then
      update public.account_security
      set must_complete_security_setup = false,
          security_setup_completed_at = coalesce(security_setup_completed_at, now()),
          updated_at = now()
      where user_id = v_uid;
      v_must_complete := false;
    else
      v_must_complete := coalesce(v_sec.must_complete_security_setup, true) and v_completed_count < 3;
    end if;
  else
    v_must_complete := false;
    v_completed_count := 3;
    v_pw_done := true;
    v_email_done := true;
    v_pin_done := true;
  end if;

  return jsonb_build_object(
    'ok', true,
    'user_id', v_uid,
    'email', v_user.email,
    'role', coalesce(v_role, 'customer'),
    'is_privileged', v_is_privileged,
    'must_complete_security_setup', v_must_complete,
    'password_completed', v_pw_done,
    'email_completed', v_email_done,
    'pin_completed', v_pin_done,
    'completed_count', v_completed_count,
    'total_steps', 3,
    'email_confirmed', v_email_confirmed,
    'password_changed_at', v_sec.password_changed_at,
    'email_verified_at', v_sec.email_verified_at,
    'pin_created_at', v_sec.pin_created_at,
    'security_setup_completed_at', v_sec.security_setup_completed_at
  );
end;
$$;

-- 8. RPC: verify_and_reset_pin (PIN recovery, no ON CONFLICT)
-- ------------------------------------------------------------------------------
create or replace function public.verify_and_reset_pin(p_token text, p_new_pin text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, auth
as $$
declare
  v_clean_token text := trim(coalesce(p_token, ''));
  v_clean_pin text := trim(coalesce(p_new_pin, ''));
  v_rec record;
  v_role text;
  v_found boolean := false;
  v_new_pin_hash text;
begin
  if v_clean_token = '' then
    return jsonb_build_object('ok', false, 'error', 'Token pemulihan PIN tidak valid.');
  end if;

  for v_rec in
    select id, user_id, email, token_hash
    from public.pin_recovery_tokens
    where used_at is null and expires_at > now()
    order by created_at desc
  loop
    if v_rec.token_hash = extensions.crypt(v_clean_token, v_rec.token_hash) then
      v_found := true;
      exit;
    end if;
  end loop;

  if not v_found then
    return jsonb_build_object('ok', false, 'error', 'Tautan pemulihan PIN tidak valid atau telah kedaluwarsa.');
  end if;

  select role into v_role from public.user_roles where user_id = v_rec.user_id limit 1;

  if v_role in ('admin', 'master_admin', 'master_helpdesk') then
    if v_clean_pin !~ '^\d{12}$' then
      return jsonb_build_object('ok', false, 'error', 'PIN wajib terdiri dari 12 angka.');
    end if;
  else
    if v_clean_pin !~ '^\d{8,12}$' then
      return jsonb_build_object('ok', false, 'error', 'PIN baru harus berupa angka antara 8 hingga 12 digit.');
    end if;
  end if;

  v_new_pin_hash := extensions.crypt(v_clean_pin, extensions.gen_salt('bf', 8));

  perform public.mb_upsert_user_security_pin(v_rec.user_id, v_rec.email, v_new_pin_hash);
  perform public.mb_ensure_account_security(v_rec.user_id, v_role in ('admin', 'master_admin', 'master_helpdesk'));

  update public.account_security
  set pin_created_at = now(),
      pin_set_at = now(),
      updated_at = now()
  where user_id = v_rec.user_id;

  update public.pin_recovery_tokens
  set used_at = now()
  where id = v_rec.id;

  return jsonb_build_object('ok', true, 'message', 'PIN telah diperbarui.');
end;
$$;

-- ------------------------------------------------------------------------------
-- 3. HELPER: mb_sync_profile
-- Guarantees a valid row with id = p_uid exists in public.profiles.
-- Cleans up any stale row with conflicting email first.
-- ------------------------------------------------------------------------------
create or replace function public.mb_sync_profile(
  p_uid uuid,
  p_email text,
  p_name text default null
)
returns void
language plpgsql
security definer
set search_path = public, auth
as $$
declare
  v_clean_email text := lower(trim(coalesce(p_email, '')));
  v_clean_name text := nullif(trim(coalesce(p_name, '')), '');
begin
  if p_uid is null then
    return;
  end if;

  -- Step 1: Free up conflicting email constraint from any stale unreferenced row
  if v_clean_email <> '' then
    update public.profiles
    set email = email || '.old.' || substr(id::text, 1, 8),
        updated_at = now()
    where lower(email) = v_clean_email
      and id <> p_uid;
  end if;

  -- Step 2: Strictly guarantee row with id = p_uid exists in public.profiles
  if exists (select 1 from public.profiles where id = p_uid) then
    update public.profiles
    set email = coalesce(nullif(v_clean_email, ''), email),
        full_name = coalesce(v_clean_name, nullif(trim(full_name), ''), 'Admin MyBengkel'),
        updated_at = now()
    where id = p_uid;
  else
    insert into public.profiles (id, email, full_name, created_at, updated_at)
    values (
      p_uid,
      coalesce(nullif(v_clean_email, ''), p_uid::text || '@mybengkel.local'),
      coalesce(v_clean_name, 'Admin MyBengkel'),
      now(),
      now()
    );
  end if;
exception
  when others then
    raise notice 'mb_sync_profile warning: %', sqlerrm;
end;
$$;

-- ------------------------------------------------------------------------------
-- 4. HELPER: mb_ensure_user_role
-- Guarantees public.profiles exists BEFORE any insert/update into public.user_roles.
-- ------------------------------------------------------------------------------
create or replace function public.mb_ensure_user_role(
  p_uid uuid,
  p_fallback_role text default 'admin'
)
returns text
language plpgsql
security definer
set search_path = public, auth
as $$
declare
  v_role text;
  v_meta_role text;
  v_email text;
  v_full_name text;
begin
  if p_uid is null then
    return null;
  end if;

  -- 1. Check existing role in user_roles
  select role into v_role
  from public.user_roles
  where user_id = p_uid
  order by assigned_at desc nulls last
  limit 1;

  if v_role is not null then
    return v_role;
  end if;

  -- 2. Read onboarding metadata from auth.users
  select email, raw_user_meta_data->>'onboarding_role', raw_user_meta_data->>'full_name'
  into v_email, v_meta_role, v_full_name
  from auth.users
  where id = p_uid;

  if v_meta_role in ('admin', 'master_admin', 'master_helpdesk', 'admin_gudang', 'master_gudang', 'mekanik', 'customer') then
    v_role := v_meta_role;
  else
    v_role := coalesce(p_fallback_role, 'admin');
  end if;

  -- 3. CRITICAL: Guarantee public.profiles row exists BEFORE touching public.user_roles
  perform public.mb_sync_profile(p_uid, v_email, v_full_name);

  -- 4. Assign role in public.user_roles
  if not exists (select 1 from public.user_roles where user_id = p_uid and role = v_role) then
    if exists (select 1 from public.user_roles where user_id = p_uid) then
      update public.user_roles set role = v_role, assigned_at = now() where user_id = p_uid;
    else
      insert into public.user_roles (user_id, role, assigned_at)
      values (p_uid, v_role, now());
    end if;
  end if;

  return v_role;
end;
$$;

-- ------------------------------------------------------------------------------
-- 5. RPC: set_account_security_pin
-- Guaranteed execution order across Admin, Master Admin, and Master Helpdesk.
-- ------------------------------------------------------------------------------
create or replace function public.set_account_security_pin(
  p_new_pin text,
  p_role text default null
)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, auth
as $$
declare
  v_user_id uuid := auth.uid();
  v_user record;
  v_email text;
  v_meta_role text;
  v_full_name text;
  v_role text;
  v_clean_pin text := trim(coalesce(p_new_pin, ''));
  v_pin_hash text;
begin
  -- Step a: Validate authentication
  if v_user_id is null then
    return jsonb_build_object('ok', false, 'error', 'Sesi tidak terautentikasi.');
  end if;

  -- Step b: Obtain user details from auth.users
  select id, email, email_confirmed_at, raw_user_meta_data
  into v_user
  from auth.users
  where id = v_user_id;

  if not found then
    return jsonb_build_object('ok', false, 'error', 'Data pengguna auth tidak ditemukan.');
  end if;

  v_email := v_user.email;
  v_meta_role := v_user.raw_user_meta_data->>'onboarding_role';
  v_full_name := v_user.raw_user_meta_data->>'full_name';

  -- Step c: Determine target role (validating admin, master_admin, master_helpdesk)
  if p_role in ('admin', 'master_admin', 'master_helpdesk', 'admin_gudang', 'master_gudang', 'mekanik', 'customer') then
    v_role := p_role;
  elsif v_meta_role in ('admin', 'master_admin', 'master_helpdesk', 'admin_gudang', 'master_gudang', 'mekanik', 'customer') then
    v_role := v_meta_role;
  else
    select role into v_role from public.user_roles where user_id = v_user_id limit 1;
    if v_role is null then
      v_role := 'admin';
    end if;
  end if;

  -- Validate 12-digit PIN for privileged roles
  if v_role in ('admin', 'master_admin', 'master_helpdesk') then
    if v_clean_pin !~ '^\d{12}$' then
      return jsonb_build_object('ok', false, 'error', 'PIN wajib terdiri dari 12 angka.');
    end if;
  else
    if v_clean_pin !~ '^\d{8,12}$' then
      return jsonb_build_object('ok', false, 'error', 'PIN harus berupa angka antara 8 hingga 12 digit.');
    end if;
  end if;

  -- Step d: CRITICAL: BEFORE touching public.user_roles, guarantee row with id = v_user_id exists in public.profiles
  -- 1) Stale row check & email rename
  if v_email is not null and v_email <> '' then
    update public.profiles
    set email = email || '.old.' || substr(id::text, 1, 8),
        updated_at = now()
    where lower(email) = lower(v_email)
      and id <> v_user_id;
  end if;

  -- 2) Strict IF EXISTS check on public.profiles
  if exists (select 1 from public.profiles where id = v_user_id) then
    update public.profiles
    set email = coalesce(v_email, email),
        full_name = coalesce(nullif(trim(full_name), ''), v_full_name, 'Admin MyBengkel'),
        updated_at = now()
    where id = v_user_id;
  else
    insert into public.profiles (id, email, full_name, created_at, updated_at)
    values (
      v_user_id,
      coalesce(v_email, v_user_id::text || '@mybengkel.local'),
      coalesce(v_full_name, 'Admin MyBengkel'),
      now(),
      now()
    );
  end if;

  -- Step e: ONLY AFTER public.profiles is guaranteed to contain id = v_user_id, assign role in public.user_roles
  if v_role is not null then
    if exists (select 1 from public.user_roles where user_id = v_user_id) then
      update public.user_roles
      set role = v_role,
          assigned_at = now()
      where user_id = v_user_id;
    else
      insert into public.user_roles (user_id, role, assigned_at)
      values (v_user_id, v_role, now());
    end if;
  end if;

  -- Step f: Save 12-digit PIN hash into public.user_security and complete security setup
  v_pin_hash := extensions.crypt(v_clean_pin, extensions.gen_salt('bf', 8));
  perform public.mb_upsert_user_security_pin(v_user_id, v_email, v_pin_hash);

  perform public.mb_ensure_account_security(v_user_id, v_role in ('admin', 'master_admin', 'master_helpdesk'));

  update public.account_security
  set pin_created_at = now(),
      pin_set_at = now(),
      password_changed_at = coalesce(password_changed_at, now()),
      email_verified_at = coalesce(email_verified_at, v_user.email_confirmed_at, now()),
      must_complete_security_setup = false,
      security_setup_completed_at = now(),
      updated_at = now()
  where user_id = v_user_id;

  return public.get_account_security_status();
end;
$$;

  v_full_name text;
  v_role text;
  v_pin_hash text;
begin

    select id
    into v_user_id
    from auth.users
    where lower(email) = v_clean_email
    order by (email_confirmed_at is not null) desc, created_at desc
    limit 1;
  end if;

  if v_user_id is null then
    select id
    into v_user_id
    from auth.users
    where email_confirmed_at is not null
      and raw_user_meta_data->>'onboarding_role' in ('admin', 'master_admin', 'master_helpdesk')
    order by email_confirmed_at desc nulls last, created_at desc
    limit 1;
  end if;

  if v_user_id is null then

      'ok', false,
      'error', 'Data pengguna auth tidak ditemukan. Silakan verifikasi email atau login kembali.'
    );
  end if;

  select id, email, email_confirmed_at, raw_user_meta_data
  into v_user
  from auth.users
  where id = v_user_id;

  if not found then
    return jsonb_build_object('ok', false, 'error', 'Data pengguna auth tidak ditemukan.');
  end if;

  v_email := coalesce(nullif(trim(v_user.email), ''), nullif(v_clean_email, ''));
  v_meta_role := v_user.raw_user_meta_data->>'onboarding_role';
  v_full_name := v_user.raw_user_meta_data->>'full_name';

  -- Step 3: Determine target role
  if v_clean_role in ('admin', 'master_admin', 'master_helpdesk', 'admin_gudang', 'master_gudang', 'mekanik', 'customer') then
    v_role := v_clean_role;
  elsif v_meta_role in ('admin', 'master_admin', 'master_helpdesk', 'admin_gudang', 'master_gudang', 'mekanik', 'customer') then
    v_role := v_meta_role;

    select role into v_role from public.user_roles where user_id = v_user_id limit 1;
    if v_role is null then
      v_role := 'admin';
    end if;
  end if;

  update public.account_security

      pin_set_at = now(),
      password_changed_at = coalesce(password_changed_at, now()),
      email_verified_at = coalesce(email_verified_at, v_user.email_confirmed_at, now()),
      must_complete_security_setup = false,
      security_setup_completed_at = now(),
      updated_at = now()
  where user_id = v_user_id;

  return jsonb_build_object(
    'ok', true,
    'user_id', v_user_id,
    'email', v_email,
    'role', coalesce(v_role, 'admin'),
    'is_privileged', v_role in ('admin', 'master_admin', 'master_helpdesk'),
    'must_complete_security_setup', false,
    'password_completed', true,
    'email_completed', true,
    'pin_completed', true,
    'completed_count', 3,
    'total_steps', 3,
    'email_confirmed', v_user.email_confirmed_at is not null
